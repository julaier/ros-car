//构造函数      pnh_局部句柄加载初始位置，位于世界中心，
MyRobot::MyRobot() : pnh_("~"), x_(0), y_(0), theta_(0) {
    loadParams();
    if (initSerial()) {
        // 开启读取线程
        boost::thread(boost::bind(&MyRobot::serialReadThread, this));//接收串口通信，并处理发过来的信息
        
        cmd_sub_ = nh_.subscribe("/cmd_vel", 10, &MyRobot::cmdVelCallback, this);//通过cmdVelCallback来向stm32发生数据包
        odom_pub_ = nh_.advertise<nav_msgs::Odometry>("odom", 50);
        last_time_ = ros::Time::now();
        ROS_INFO("Robot Driver Core Module Loaded.");
    }
}

// 析构函数
MyRobot::~MyRobot() {
    if (serial_ptr_) serial_ptr_->close();
}

//通过pnh来获取参数
void MyRobot::loadParams() {
    pnh_.param<std::string>("robot_port", port_name_, "/dev/ttyAMA1");//为AMA0
    pnh_.param<int>("baud_rate", baud_rate_, 115200);//波特率
    pnh_.param<std::string>("odom_frame", odom_frame_, "odom");
    pnh_.param<std::string>("base_frame", base_frame_, "base_footprint");
}

bool MyRobot::initSerial() {
    try {
        serial_ptr_ = std::make_shared<boost::asio::serial_port>(io_service_);
        serial_ptr_->open(port_name_);
        serial_ptr_->set_option(boost::asio::serial_port_base::baud_rate(baud_rate_));
        return true;
    } catch (std::exception& e) {
        ROS_ERROR("Serial Error: %s", e.what());
        return false;
    }
}

//循环读取串口的线程
void MyRobot::serialReadThread() {
    // 1. 准备一个较大的临时接收缓冲区
    uint8_t temp_buf[1024]; 
    uint8_t rx_buf[64]; // 用于存放正在解析的单包数据
    uint8_t state = 0;
    uint8_t rx_num = 0;
    uint8_t data_len = 0;

    while (ros::ok()) {
        try {
            // 2. 批量读取：不再是 1 字节，而是尽可能多读（read_some 是非阻塞或部分阻塞的）
            // 它会等待至少有 1 个字节可用，然后一次性把缓冲区里的数据全拿出来
            size_t bytes_read = serial_ptr_->read_some(boost::asio::buffer(temp_buf, sizeof(temp_buf)));

            if (bytes_read > 0) {
                // 3. 在内存中循环处理这一批数据，不再触发系统调用
                for (size_t i = 0; i < bytes_read; ++i) {
                    uint8_t res = temp_buf[i];

                    // --- 状态机逻辑开始 (保持不变，但运行在内存循环中) ---
                    if (state == 0) {
                        if (res == HEADER_1) { 
                            rx_buf[rx_num++] = res; state = 1; 
                        }
                    }
                    else if (state == 1) {
                        if (res == HEADER_2) { 
                            rx_buf[rx_num++] = res; state = 2; 
                        } else { 
                            state = 0; rx_num = 0; // 头不对，重置
                        }
                    }
                    else if (state == 2) { 
                        rx_buf[rx_num++] = res; state = 3; // ID
                    }
                    else if (state == 3) { 
                        rx_buf[rx_num++] = res; data_len = res; state = 4; // Length
                    }
                    else if (state == 4) {
                        rx_buf[rx_num++] = res;
                        if (rx_num == data_len + 4) state = 5;
                    }
                    else if (state == 5) { // Checksum
                        if (ProtocolHelper::calculateChecksum(rx_buf, rx_num) == res) {
                            processFeedback(rx_buf);
                        }
                        state = 0; rx_num = 0; // 处理完一包，重置状态
                    }
                    // --- 状态机逻辑结束 ---

                    // 安全检查：防止 rx_buf 溢出
                    if (rx_num >= sizeof(rx_buf)) {
                        state = 0; rx_num = 0;
                    }
                }
            }
        } catch (std::exception& e) {
            ROS_ERROR("Serial Read Thread Error: %s", e.what());
            // 如果串口断开，尝试等待并重连或报错
            ros::Duration(1.0).sleep();
        }
    }
}
// 回调函数 //树莓派->stm32发送数据，构造数据包
void MyRobot::cmdVelCallback(const geometry_msgs::Twist::ConstPtr& msg) {
    uint8_t tx_buf[32]; 
    ControlData data;
    data.vx = (int16_t)(msg->linear.x * 1000);

    data.vy = (int16_t)(msg->linear.y *1000); //横移速度
    data.wz = (int16_t)(msg->angular.z * 1000);
    
    double move_direction = std::atan2(msg->linear.y,msg->linear.x);
    data.yaw = (int16_t) (move_direction * 1000);

    tx_buf[0] = HEADER_1; tx_buf[1] = HEADER_2; tx_buf[2] = ID_ROS2CTR_VEL; tx_buf[3] = (uint16_t)sizeof(data);
    memcpy(&tx_buf[4], &data, sizeof(data));
    tx_buf[4 + sizeof(data)] = ProtocolHelper::calculateChecksum(tx_buf, 4 + sizeof(data));
    try
    {
    boost::asio::write(*serial_ptr_, boost::asio::buffer(tx_buf, 5 + sizeof(data)));
    }
    catch (boost::system::system_error& e) {
        ROS_ERROR("Failed to write to serial port: %s", e.what());
    }
}

void MyRobot::processFeedback(uint8_t* buf) {
    if (buf[2] == ID_CTR2ROS_DATA) {
        double vx = (int16_t)(buf[5] << 8 | buf[4]) / 1000.0; //第三个字节
        double vy = (int16_t)(buf[7] << 8 |buf[6])/1000.0;     //第四个字节
        double wz = (int16_t)(buf[9] << 8 | buf[8]) / 1000.0;  //第5个字节
        double yaw = (int16_t)(buf[11] << 8 | buf[10]) / 1000.0; //第六个字节 
        publishOdomAndTF(vx,vy, wz, yaw);
    }
}

void MyRobot::publishOdomAndTF(double vx, double vy,double wz, double yaw) {
    ros::Time now = ros::Time::now();
    
    if(last_time_.isZero())
    {
        last_time_ = now;
        return;
    }
    double dt = (now - last_time_).toSec();
    // 利用航向角 将机器人本地速度旋转到全局坐标坐标系

    //里程计计算的公式解析  通过旋转变换矩阵
        //小车坐标系转到世界坐标系，为什么要结合theta来计算世界坐标系下的位移
        //因为如果yaw变了的话，如果是向斜向上运动，在小车的坐标系下是直走，
        //但是实际世界坐标系是yaw和vx结合计算得到
        // theta_ = yaw
    /*
    [x global]    =  [cos(theta_)   -sin(theta_)]     [vxlocal]
    [y global]       [sin(theta_)    cos(theta_)]     [vylocal]
    
    */


    //瞬时位移
    // 假设上一个时刻的角度是 prev_theta
    double avg_theta = theta_ + (wz * dt) / 2.0; 
    double delta_x = (vx * cos(avg_theta) - vy * sin(avg_theta)) * dt;
    double delta_y = (vx * sin(avg_theta) + vy * cos(avg_theta)) * dt;
 
    //积分，更新xy值
    x_ += delta_x;
    y_ += delta_y;
    theta_ = yaw; // 直接采用 IMU 修正后的航向角

    tf2::Quaternion q; q.setRPY(0, 0, theta_);//设置翻转，俯仰，偏航，转为四元数
    geometry_msgs::TransformStamped tf_msg; // TF数据包
    tf_msg.header.stamp = now; //时间戳
    tf_msg.header.frame_id = odom_frame_; tf_msg.child_frame_id = base_frame_;   //发布坐标系从odom ->base_frame_;
    tf_msg.transform.translation.x = x_; //x位移
    tf_msg.transform.translation.y = y_; //y位移
    tf_msg.transform.translation.z = 0.0;
    tf_msg.transform.rotation = tf2::toMsg(q);//四元数
    tf_broadcaster_.sendTransform(tf_msg);//发送


    // 发布导航信息
    nav_msgs::Odometry odom;
    odom.header = tf_msg.header; odom.child_frame_id = base_frame_;
    odom.pose.pose.position.x = x_; odom.pose.pose.position.y = y_; odom.pose.pose.position.z = 0.0;
    odom.pose.pose.orientation = tf_msg.transform.rotation;
    odom.twist.twist.linear.x = vx; 
    odom.twist.twist.linear.y = vy;
    odom.twist.twist.angular.z = wz;
    
    //协方差
    //0:x, 7:y, 35:yaw (6x6 矩阵的对角线)
    odom.pose.covariance[0]  = 1e-3;  // X轴
    odom.pose.covariance[7]  = 1e-2;  // Y轴（麦轮横移误差大一些）
    odom.pose.covariance[35] = 1e-3;  // Ya
    last_time_ = now;
}
